# 创建组件

## 目录结构

```bash
src/components/              # 全局组件
├── os-your-component/       # 组件根目录（kebab-case）
│   ├── components/          # 私有组件
│   │   ├── user-filter.ts
│   │   └── user-filter.vue
│   ├── styles/              # 样式
│   │   ├── index.scss
│   │   └── user-filter.scss
│   ├── index.vue            # 组件主体
│   └── types.ts             # 类型定义
```

## 代码示例

### types.ts

```typescript
export interface YourComponentProps {
  disabled?: boolean
}

export interface YourComponentEmits {
  click: [event: MouseEvent]
}
```

### index.vue

```vue
<script setup lang="ts">
import { YourComponentProps, YourComponentEmits } from './types'

defineOptions({
  name: 'OsYourComponent',
})

const props = withDefaults(defineProps<YourComponentProps>(), {
  // 必传的属性不需要定义默认值，只需要定义类型即可
  disabled: false,
})
const emit = defineEmits<YourComponentEmits>()

const handleClick = (e: MouseEvent) => {
  emit('click', e)
}
</script>

<template>
  <div class="os-your-component">
    <!-- 组件内容 -->
  </div>
</template>

<style lang="scss" scoped>
@import './style/index.scss';
</style>
```
